"""Fix schema $id format per ADR-012 + naming.md.
Pattern: http://platinum-seo-engine/schemas/<slug>

Usage:
  python3 scripts/maintenance/fix_schema_id_format.py --dry-run
  python3 scripts/maintenance/fix_schema_id_format.py --apply
"""
import argparse
import json
import re
from pathlib import Path

SCHEMAS_DIR = Path(__file__).parent.parent.parent / "schemas"
SLUG_FROM_FILENAME = re.compile(r"^([a-z][a-z0-9-]*)\.(?:schema\.)?json$")


def expected_id(filename: str) -> str:
    m = SLUG_FROM_FILENAME.match(filename)
    if not m:
        raise ValueError(f"Unexpected filename: {filename}")
    return f"http://platinum-seo-engine/schemas/{m.group(1)}"


def fix_one(schema_path: Path, apply: bool) -> tuple[str, str | None, str | None]:
    with schema_path.open() as f:
        data = json.load(f)
    current = data.get("$id")
    expected = expected_id(schema_path.name)
    if current == expected:
        return ("ok", current, expected)
    if apply:
        data["$id"] = expected
        with schema_path.open("w") as f:
            json.dump(data, f, indent=2, ensure_ascii=False)
            f.write("\n")
    return ("change", current, expected)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--apply", action="store_true")
    parser.add_argument("--dry-run", action="store_true")
    args = parser.parse_args()
    if not (args.apply or args.dry_run):
        parser.error("--apply or --dry-run required")

    changed = 0
    for schema_path in sorted(SCHEMAS_DIR.glob("*.json")):
        try:
            status, current, expected = fix_one(schema_path, apply=args.apply)
        except ValueError as e:
            print(f"SKIP {schema_path.name}: {e}")
            continue
        if status == "change":
            changed += 1
            verb = "APPLY" if args.apply else "DRY-RUN"
            print(f"{verb} {schema_path.name}: {current!r} → {expected!r}")
    print(f"\nTotal: {changed} schema(s) need(s) fix.")


if __name__ == "__main__":
    main()
```

"""Regression: cross-schema enum drift detection.
events.schema.json work.primary_source MUST be ⊆ master-excel.schema.json master_task.primary_source.
"""
import json
from pathlib import Path

SCHEMAS = Path(__file__).parent.parent.parent / "schemas"

def _events_primary_source():
    with (SCHEMAS / "events.schema.json").open() as f:
        d = json.load(f)
    # Walk to work event primary_source enum
    return set(d["properties"]["primary_source"]["enum"]) if "primary_source" in d["properties"] else \
           _find_primary_source_enum(d)

def _find_primary_source_enum(obj, depth=0):
    if depth > 20: return set()
    if isinstance(obj, dict):
        if "primary_source" in obj and "enum" in obj.get("primary_source", {}):
            return set(obj["primary_source"]["enum"])
        for v in obj.values():
            r = _find_primary_source_enum(v, depth+1)
            if r: return r
    elif isinstance(obj, list):
        for v in obj:
            r = _find_primary_source_enum(v, depth+1)
            if r: return r
    return set()

def _master_primary_source():
    with (SCHEMAS / "master-excel.schema.json").open() as f:
        d = json.load(f)
    cols = d["properties"]["sheets"]["properties"]["master_task"]["properties"]["required_columns"]
    ps_col = next(c for c in cols if c.get("name") == "primary_source")
    return set(ps_col["enum"])

def test_events_primary_source_subset_of_master():
    events = _events_primary_source()
    master = _master_primary_source()
    missing = master - events
    assert not missing, (
        f"events.schema primary_source missing values present in master-excel: {missing}. "
        f"Per ADR-018 additive policy, events MUST mirror master superset."
    )

"""Regression: every rules/*.md MUST have valid frontmatter per rules-frontmatter.schema.json."""
import re
from pathlib import Path
import json
import jsonschema
import pytest
import yaml

RULES_DIR = Path(__file__).parent.parent.parent / "rules"
SCHEMA_PATH = Path(__file__).parent.parent.parent / "schemas" / "rules-frontmatter.schema.json"
FRONTMATTER_RE = re.compile(r"^---\n(.*?)\n---\n", re.DOTALL)


def _load_frontmatter(rule_path):
    text = rule_path.read_text()
    m = FRONTMATTER_RE.match(text)
    if not m:
        return None
    return yaml.safe_load(m.group(1))


@pytest.mark.parametrize("rule_path", sorted(RULES_DIR.glob("*.md")))
def test_rule_has_frontmatter(rule_path):
    fm = _load_frontmatter(rule_path)
    assert fm is not None, f"{rule_path.name}: no frontmatter"


@pytest.mark.parametrize("rule_path", sorted(RULES_DIR.glob("*.md")))
def test_rule_frontmatter_valid(rule_path):
    fm = _load_frontmatter(rule_path)
    if fm is None:
        pytest.skip("no frontmatter")
    schema = json.loads(SCHEMA_PATH.read_text())
    jsonschema.validate(fm, schema)
```

"""Regression: every rule that claims a pre-commit hook MUST have the script."""
from pathlib import Path
import re
import pytest

ROOT = Path(__file__).parent.parent.parent
RULES = ROOT / "rules"
HOOK_REF_RE = re.compile(r"scripts/hooks/(\w+\.(?:sh|py))")

def test_all_hook_refs_exist():
    missing = []
    for rule in RULES.glob("*.md"):
        for m in HOOK_REF_RE.finditer(rule.read_text()):
            script = ROOT / "scripts" / "hooks" / m.group(1)
            if not script.exists():
                missing.append(f"{rule.name} → scripts/hooks/{m.group(1)}")
    assert not missing, f"Missing hook scripts: {missing}"
